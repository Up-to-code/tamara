  function detectTypes() {
    const detected = [];
    for (const [type, detector] of Object.entries(detectionTypes)) {
      if (detector()) {
        detected.push(type);
      }
    }
    return detected;
  }

  function evaluateCondition(condition, currentTypes) {
    const parts = condition.split('&').map((part) => part.trim());
    return parts.every((part) => {
      const orParts = part.split('|').map((p) => p.trim());
      return orParts.some((p) => {
        if (p.startsWith('!')) {
          return !currentTypes.includes(p.slice(1));
        }
        return currentTypes.includes(p);
      });
    });
  }

/**
   * Persist & Enforce UTM across pages in Webflow (SESSION ONLY)
   * - Captures UTM params on first visit of the SESSION and stores them in sessionStorage
   * - Internal links: always overwrite UTM with stored session UTMs
   * - Outbound links (to defined domains): remove all original UTMs and replace with session UTMs
   * - Non-UTM query params and hash fragments are preserved
   *
   * IMPORTANT: UTMs DO NOT persist across browser restarts (sessionStorage).
   * Legacy cleanup: removes any old localStorage key to avoid reviving stale UTMs.
   *
   * Opt-out: add data-utm-keep="false" to <a> tag if you don’t want UTM appended
   */

  (function () {
    // ---- Config ----
    var UTM_KEYS = ['utm_source', 'utm_medium', 'utm_campaign', 'utm_term', 'utm_content', 'gclid', 'fbclid'];

    // Session-scoped storage key
    var SESSION_STORAGE_KEY = 'utm_params_session';

    // Optional: remove legacy localStorage value left by previous version
    var CLEAR_LEGACY_LOCALSTORAGE = true;

    // Outbound domains to enforce UTMs on (extend as needed)
    var OUTBOUND_DOMAINS = ['https://business.tamara.co', 'https://partners.tamara.co'];

    var DEBUG = false;
    function log() {
      if (DEBUG) {
        try {
          console.log.apply(console, arguments);
        } catch (e) {}
      }
    }

    // --- Storage helpers (SESSION ONLY) ---
    function loadSessionUtm() {
      try {
        var raw = sessionStorage.getItem(SESSION_STORAGE_KEY);
        if (!raw) return {};
        return JSON.parse(raw) || {};
      } catch (e) {
        return {};
      }
    }
    function saveSessionUtm(obj) {
      try {
        sessionStorage.setItem(SESSION_STORAGE_KEY, JSON.stringify(obj || {}));
      } catch (e) {}
    }

    // Legacy cleanup (one-way)
    (function cleanupLegacy() {
      if (!CLEAR_LEGACY_LOCALSTORAGE) return;
      try {
        localStorage.removeItem('utm_params');
      } catch (e) {}
    })();

    // Read UTM from current URL (whitelist keys)
    function readUtmFromLocation() {
      var params = new URLSearchParams(window.location.search);
      var out = {};
      UTM_KEYS.forEach(function (k) {
        var v = params.get(k);
        if (v) out[k] = v;
      });
      return out;
    }

    // Merge: new overrides old (for mid-session updates)
    function mergeUtm(oldUtm, newUtm) {
      var merged = Object.assign({}, oldUtm);
      Object.keys(newUtm || {}).forEach(function (k) {
        if (newUtm[k]) merged[k] = newUtm[k];
      });
      return merged;
    }

    // Internal link check
    function isInternalLink(a) {
      var href = a.getAttribute('href');
      if (!href) return false;
      if (href.startsWith('#')) return false;
      if (href.startsWith('mailto:')) return false;
      if (href.startsWith('tel:')) return false;
      if (href.startsWith('javascript:')) return false;
      if (href.startsWith('/')) return true;
      try {
        var url = new URL(href, window.location.origin);
        return url.origin === window.location.origin;
      } catch (e) {
        return false;
      }
    }

    // Internal links: overwrite UTM params with session UTMs
    function withUtmOverwriteInternal(href, utmObj) {
      if (!utmObj || Object.keys(utmObj).length === 0) return href;

      var base = new URL(href, window.location.origin);
      var qs = new URLSearchParams(base.search);

      UTM_KEYS.forEach(function (k) {
        if (utmObj[k]) qs.set(k, utmObj[k]);
        else qs.delete(k); // optional: remove empty UTMs if present
      });

      base.search = qs.toString() ? '?' + qs.toString() : '';
      return base.pathname + base.search + base.hash;
    }

    // Outbound links: replace ALL UTMs with the session set (preserve non-UTM params)
    function withUtmReplaceAllOutbound(href, utmObj) {
      if (!utmObj || Object.keys(utmObj).length === 0) return href;

      var base = new URL(href);
      var qs = new URLSearchParams(base.search);

      // Remove all existing UTM keys
      UTM_KEYS.forEach(function (k) {
        qs.delete(k);
      });

      // Add stored session UTMs
      Object.keys(utmObj).forEach(function (k) {
        qs.set(k, utmObj[k]);
      });

      base.search = qs.toString() ? '?' + qs.toString() : '';
      return base.origin + base.pathname + base.search + base.hash;
    }

    // Apply UTMs to all links on the page
    function applyUtmToLinks(utmObj) {
      if (!utmObj || Object.keys(utmObj).length === 0) return;

      var links = document.querySelectorAll('a[href]');
      links.forEach(function (a) {
        var keep = a.getAttribute('data-utm-keep');
        if (keep && keep.toLowerCase() === 'false') return;

        var orig = a.getAttribute('href');

        // Outbound domains
        var outboundMatch = OUTBOUND_DOMAINS.find(function (d) {
          return orig.startsWith(d);
        });

        if (outboundMatch) {
          var newHref = withUtmReplaceAllOutbound(orig, utmObj);
          if (newHref !== orig) {
            a.setAttribute('href', newHref);
            log('UTM outbound (session):', { from: orig, to: newHref });
          }
          return;
        }

        // Internal
        if (isInternalLink(a)) {
          var newHref = withUtmOverwriteInternal(orig, utmObj);
          if (newHref !== orig) {
            a.setAttribute('href', newHref);
            log('UTM internal (session):', { from: orig, to: newHref });
          }
        }
      });
    }

    // Capture & persist UTMs FOR THIS SESSION ONLY
    function captureAndPersistSession() {
      var current = readUtmFromLocation();
      if (Object.keys(current).length === 0) return; // no UTM on this page => do NOT create or revive any
      var stored = loadSessionUtm();
      var merged = mergeUtm(stored, current); // current overrides stored
      saveSessionUtm(merged);
      log('UTM stored in session:', merged);
    }

    // Observe DOM (Webflow dynamic injections)
    function setupObserver(utmObj) {
      if (!window.MutationObserver) return;
      var obs = new MutationObserver(function () {
        applyUtmToLinks(utmObj);
      });
      obs.observe(document.documentElement, { childList: true, subtree: true });
    }

    function init() {
      // 1) Only store UTMs if present in CURRENT URL (session-scoped)
      captureAndPersistSession();

      // 2) Load session UTMs (if any) and apply to links
      var sessionUtm = loadSessionUtm();
      applyUtmToLinks(sessionUtm);

      // 3) Keep applying on DOM changes (for late-added links)
      setupObserver(sessionUtm);
    }

    if (document.readyState === 'loading') {
      __tlReady( init);
    } else {
      init();
    }
  })();

__tlReady( () => {
    const isMobile = window.matchMedia('(max-width: 767px)').matches;

    // Single shared observer for all items
    const observer = new IntersectionObserver(
      (entries) => {
        entries.forEach((entry) => {
          if (entry.isIntersecting) {

            observer.unobserve(entry.target);
          }
        });
      },
      {
        threshold: 0,
        rootMargin: '0px 0px -10% 0px',
      },
    );

    // Set initial hidden state for each item
    const initItem = (item, delay = 0) => {
      item.style.opacity = '0';
      item.style.transform = 'translateY(24px)';
      item.style.willChange = 'opacity, transform';
      item.style.transition = 'opacity 1.2s ease, transform 1s cubic-bezier(0.22, 1, 0.36, 1)';
      item.style.transitionDelay = isMobile ? '0ms' : `${delay}ms`;
    };

    // Trigger fade-up animation and free GPU memory after transition
    const showItem = (item) => {
      item.style.opacity = '1';
      item.style.transform = 'translateY(0)';
      item.addEventListener(
        'transitionend',
        () => {
          item.style.willChange = 'auto';
        },
        { once: true },
      );
    };

    // Auto-apply fade-up classes to FAQ section
    document.querySelectorAll('.la-component_faq').forEach((faq) => {
      faq.classList.add('fade-up-group');
      faq.querySelectorAll('.la-faq-dropdown-item').forEach((item) => {
        item.classList.add('fade-up-item');
      });
    });

    // Items inside .fade-up-group — apply stagger delay on desktop, capped at 400ms
    document.querySelectorAll('.fade-up-group').forEach((group) => {
      group.querySelectorAll('.fade-up-item').forEach((item, index) => {
        initItem(item, Math.min(index * 150, 400));
        observer.observe(item);
      });
    });

    // Items outside .fade-up-group — observe individually with no stagger
    document.querySelectorAll('.fade-up-item').forEach((item) => {
      if (!item.closest('.fade-up-group')) {
        initItem(item);
        observer.observe(item);
      }
    });

__tlReady( function () {
    const buttonIds = ["body-btn-shop", "hero-btn-shop"];

    buttonIds.forEach(id => {
      const link = document.getElementById(id);
      if (!link) return;

      const href = link.getAttribute("href");
      if (!href) return;

      // Skip nếu đã có đủ UTM
      if (
        href.includes("utm_source=Tamara") &&
        href.includes("utm_medium=Web_Directory")
      ) {
        return;
      }

      // Skip mailto, tel, anchor
      if (
        href.startsWith("mailto:") ||
        href.startsWith("tel:") ||
        href.startsWith("#")
      ) {
        return;
      }

      let url;

      try {
        url = new URL(href, window.location.origin);
      } catch (e) {
        return;
      }

      // Add UTM nếu chưa có
      if (!url.searchParams.has("utm_source")) {
        url.searchParams.append("utm_source", "Tamara");
      }
      if (!url.searchParams.has("utm_medium")) {
        url.searchParams.append("utm_medium", "Web_Directory");
      }

      link.setAttribute("href", url.toString());
    });
