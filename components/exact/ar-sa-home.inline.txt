    function isElementRendered(element) {
      if (
        !element ||
        element.getClientRects().length === 0

        return false;
      }

      var style = window.getComputedStyle(element);

      return (
        style.display !== 'none' &&

    function setVideoTime(video, time) {
      try {
        video.currentTime = Math.max(0, time);
      } catch (error) {
        console.warn(
          'Unable to update motion video time:',
          error
        );

     * One-time setup shared by every video, including
     * the locale variants that are currently hidden.

        return;
      }

      video.dataset.videoController = 'motion-hover';

      video.muted = true;
      video.defaultMuted = true;
      video.playsInline = true;
      video.autoplay = false;
      video.loop = false;

      video.setAttribute('muted', '');
      video.setAttribute('playsinline', '');
      video.setAttribute('webkit-playsinline', '');
      video.setAttribute('preload', 'auto');

      video.removeAttribute('loop');

          video.loop = false;
          video.removeAttribute('loop');

var Webflow = Webflow || [];

  Webflow.push(function () {
    var api = Webflow.require('lottie');

    if (!api || typeof api.createInstance !== 'function') {
      console.warn('Webflow lottie API is unavailable');
      return;
    }

    var TARGET = '.lottie-scroll-to-play-v1';
    var PLAY_RATIO = 0.9;
    var TICK_MS = 100;
    var MAX_WAIT_MS = 8000;

    var els = Array.prototype.slice.call(document.querySelectorAll(TARGET));

    if (!els.length) {
      return;
    }

    var desired = new WeakMap();
    var played = new WeakMap();
    var timers = new WeakMap();

    /*
     * Webflow re-reads data-autoplay on every load(), and
     * createInstance() calls load() again. While the attribute
     * exists, Webflow starts the animation from frame 0 by itself:
     * once during lazy load (offscreen, at 150% of the viewport)
     * and again through its own visibility observer, which has no
     * threshold. Removing the attribute is the only reliable way
     * to keep playback under this script's control.
     */
    function disableWebflowAutoplay(el) {
      el.removeAttribute('data-autoplay');
    }

    function instanceFor(el) {
      var inst;

      try {
        inst = api.createInstance(el);
      } catch (error) {
        console.warn('Unable to create lottie instance for:', el, error);
        return null;
      }

      if (inst) {
        /*
         * Playback Webflow has already queued internally.
         * Left set, its visibility observer restarts or resumes
         * the animation on the next scroll.
         */
        inst._pendingAutoplay = false;
        inst._wasPlayingBeforePause = false;
      }

      return inst;
    }

    /*
     * Returns true once the wanted state is confirmed, which ends
     * polling. Webflow destroys and recreates the instance the first
     * time the element intersects, wiping its event handlers, so the
     * instance is re-resolved on every attempt instead of listening
     * for dataReady once.
     */
    function applyState(el) {
      var inst = instanceFor(el);

      if (!inst || !inst.isLoaded) {
        return false;
      }

      var item = inst.animationItem;

      if (!item) {
        return false;
      }

      /*
       * Set directly. Webflow's setLooping() calls stop() while
       * playing, which visibly jumps back to frame 0.
       */
      item.loop = false;

      if (desired.get(el) !== 'play') {
        item.goToAndStop(0, true);
        return true;
      }

      if (item.isPaused && item.currentFrame <= 0) {
        item.goToAndPlay(0, true);
      }

      /*
       * Only a frame past 0 proves playback really started on the
       * instance that survived Webflow's recreate step.
       */
      return item.currentFrame > 0;
    }

    function poll(el) {
      if (timers.get(el)) {
        return;
      }

      var waited = 0;

      var timer = window.setInterval(function () {
        waited += TICK_MS;

        if (applyState(el) || waited >= MAX_WAIT_MS) {
          window.clearInterval(timer);
          timers.set(el, null);
        }
      }, TICK_MS);

      timers.set(el, timer);
    }

    function request(el, state) {
      desired.set(el, state);

      if (!applyState(el)) {
        poll(el);
      }
    }

    function viewportHeight() {
      return window.innerHeight || document.documentElement.clientHeight;
    }

    /*
     * An element taller than the viewport can never reach 90%
     * visibility, so cap the requirement at what is achievable.
     */
    function requiredRatio(el) {
      var height = el.getBoundingClientRect().height;

      if (!height) {
        return PLAY_RATIO;
      }

      return Math.min(PLAY_RATIO, Math.min(1, viewportHeight() / height) * 0.9);
    }

    /*
     * Above-the-fold animations must play on page load. They are
     * often cropped by the bottom of the viewport, so they would
     * never reach the scroll threshold and would sit on frame 0,
     * which reads as a missing visual.
     */
    function isInInitialViewport(el) {
      var rect = el.getBoundingClientRect();

      return rect.top < viewportHeight() && rect.bottom > 0;
    }

    var io = new IntersectionObserver(
      function (entries) {
        entries.forEach(function (entry) {
          var el = entry.target;

          if (entry.intersectionRatio >= requiredRatio(el)) {
            if (!played.get(el)) {
              played.set(el, true);
              request(el, 'play');
            }
            return;
          }

          /*
           * Reset only after the element has fully left the viewport.
           * Unlocking on partial leave replays the animation while
           * the user is still scrolling through the same section.
           */
          if (!entry.isIntersecting || entry.intersectionRatio <= 0) {
            if (played.get(el)) {
              played.set(el, false);
              request(el, 'stop');
            }
          }
        });
      },
      {
        threshold: [0, 0.25, 0.5, 0.7, 0.9, 1]
      }
    );

    els.forEach(function (el) {
      disableWebflowAutoplay(el);

      if (isInInitialViewport(el)) {
        played.set(el, true);
        desired.set(el, 'play');

        /*
         * Polling starts on the next tick on purpose. Creating the
         * instance now would download the JSON, then Webflow's lazy
         * pass would destroy it and download it again.
         */
        poll(el);
      } else {
        played.set(el, false);
      }

      io.observe(el);
    });
  });

/**
   * Persist & Enforce UTM across pages in Webflow (SESSION ONLY)
   * - Captures UTM params on first visit of the SESSION and stores them in sessionStorage
   * - Internal links: always overwrite UTM with stored session UTMs
   * - Outbound links (to defined domains): remove all original UTMs and replace with session UTMs
   * - Non-UTM query params and hash fragments are preserved
   *
   * IMPORTANT: UTMs DO NOT persist across browser restarts (sessionStorage).
   * Legacy cleanup: removes any old localStorage key to avoid reviving stale UTMs.
   *
   * Opt-out: add data-utm-keep="false" to <a> tag if you don’t want UTM appended
   */

  (function () {
    // ---- Config ----
    var UTM_KEYS = ['utm_source', 'utm_medium', 'utm_campaign', 'utm_term', 'utm_content', 'gclid', 'fbclid'];

    // Session-scoped storage key
    var SESSION_STORAGE_KEY = 'utm_params_session';

    // Optional: remove legacy localStorage value left by previous version
    var CLEAR_LEGACY_LOCALSTORAGE = true;

    // Outbound domains to enforce UTMs on (extend as needed)
    var OUTBOUND_DOMAINS = ['https://business.tamara.co', 'https://partners.tamara.co'];

    var DEBUG = false;
    function log() {
      if (DEBUG) {
        try {
          console.log.apply(console, arguments);
        } catch (e) {}
      }
    }

    // --- Storage helpers (SESSION ONLY) ---
    function loadSessionUtm() {
      try {
        var raw = sessionStorage.getItem(SESSION_STORAGE_KEY);
        if (!raw) return {};
        return JSON.parse(raw) || {};
      } catch (e) {
        return {};
      }
    }
    function saveSessionUtm(obj) {
      try {
        sessionStorage.setItem(SESSION_STORAGE_KEY, JSON.stringify(obj || {}));
      } catch (e) {}
    }

    // Legacy cleanup (one-way)
    (function cleanupLegacy() {
      if (!CLEAR_LEGACY_LOCALSTORAGE) return;
      try {
        localStorage.removeItem('utm_params');
      } catch (e) {}
    })();

    // Read UTM from current URL (whitelist keys)
    function readUtmFromLocation() {
      var params = new URLSearchParams(window.location.search);
      var out = {};
      UTM_KEYS.forEach(function (k) {
        var v = params.get(k);
        if (v) out[k] = v;
      });
      return out;
    }

    // Merge: new overrides old (for mid-session updates)
    function mergeUtm(oldUtm, newUtm) {
      var merged = Object.assign({}, oldUtm);
      Object.keys(newUtm || {}).forEach(function (k) {
        if (newUtm[k]) merged[k] = newUtm[k];
      });
      return merged;
    }

    // Internal link check
    function isInternalLink(a) {
      var href = a.getAttribute('href');
      if (!href) return false;
      if (href.startsWith('#')) return false;
      if (href.startsWith('mailto:')) return false;
      if (href.startsWith('tel:')) return false;
      if (href.startsWith('javascript:')) return false;
      if (href.startsWith('/')) return true;
      try {
        var url = new URL(href, window.location.origin);
        return url.origin === window.location.origin;
      } catch (e) {
        return false;
      }
    }

    // Internal links: overwrite UTM params with session UTMs
    function withUtmOverwriteInternal(href, utmObj) {
      if (!utmObj || Object.keys(utmObj).length === 0) return href;

      var base = new URL(href, window.location.origin);
      var qs = new URLSearchParams(base.search);

      UTM_KEYS.forEach(function (k) {
        if (utmObj[k]) qs.set(k, utmObj[k]);
        else qs.delete(k); // optional: remove empty UTMs if present
      });

      base.search = qs.toString() ? '?' + qs.toString() : '';
      return base.pathname + base.search + base.hash;
    }

    // Outbound links: replace ALL UTMs with the session set (preserve non-UTM params)
    function withUtmReplaceAllOutbound(href, utmObj) {
      if (!utmObj || Object.keys(utmObj).length === 0) return href;

      var base = new URL(href);
      var qs = new URLSearchParams(base.search);

      // Remove all existing UTM keys
      UTM_KEYS.forEach(function (k) {
        qs.delete(k);
      });

      // Add stored session UTMs
      Object.keys(utmObj).forEach(function (k) {
        qs.set(k, utmObj[k]);
      });

      base.search = qs.toString() ? '?' + qs.toString() : '';
      return base.origin + base.pathname + base.search + base.hash;
    }

    // Apply UTMs to all links on the page
    function applyUtmToLinks(utmObj) {
      if (!utmObj || Object.keys(utmObj).length === 0) return;

      var links = document.querySelectorAll('a[href]');
      links.forEach(function (a) {
        var keep = a.getAttribute('data-utm-keep');
        if (keep && keep.toLowerCase() === 'false') return;

        var orig = a.getAttribute('href');

        // Outbound domains
        var outboundMatch = OUTBOUND_DOMAINS.find(function (d) {
          return orig.startsWith(d);
        });

        if (outboundMatch) {
          var newHref = withUtmReplaceAllOutbound(orig, utmObj);
          if (newHref !== orig) {
            a.setAttribute('href', newHref);
            log('UTM outbound (session):', { from: orig, to: newHref });
          }
          return;
        }

        // Internal
        if (isInternalLink(a)) {
          var newHref = withUtmOverwriteInternal(orig, utmObj);
          if (newHref !== orig) {
            a.setAttribute('href', newHref);
            log('UTM internal (session):', { from: orig, to: newHref });
          }
        }
      });
    }

    // Capture & persist UTMs FOR THIS SESSION ONLY
    function captureAndPersistSession() {
      var current = readUtmFromLocation();
      if (Object.keys(current).length === 0) return; // no UTM on this page => do NOT create or revive any
      var stored = loadSessionUtm();
      var merged = mergeUtm(stored, current); // current overrides stored
      saveSessionUtm(merged);
      log('UTM stored in session:', merged);
    }

    // Observe DOM (Webflow dynamic injections)
    function setupObserver(utmObj) {
      if (!window.MutationObserver) return;
      var obs = new MutationObserver(function () {
        applyUtmToLinks(utmObj);
      });
      obs.observe(document.documentElement, { childList: true, subtree: true });
    }

    function init() {
      // 1) Only store UTMs if present in CURRENT URL (session-scoped)
      captureAndPersistSession();

      // 2) Load session UTMs (if any) and apply to links
      var sessionUtm = loadSessionUtm();
      applyUtmToLinks(sessionUtm);

      // 3) Keep applying on DOM changes (for late-added links)
      setupObserver(sessionUtm);
    }

    if (document.readyState === 'loading') {
      __tlReady( init);
    } else {
      init();
    }
  })();
